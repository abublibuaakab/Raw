local function getVehicle()
    local char = LocalPlayer.Character
    if char and char:FindFirstChild("Humanoid") and char.Humanoid.SeatPart then

RunService.RenderStepped:Connect(function()
    local currentVehicle = getVehicle()
    FlyGui.Enabled = (currentVehicle ~= nil and flyEnabled)
    if not flyEnabled or not bv or not bg or not currentVehicle then return end
    local root = currentVehicle.PrimaryPart or currentVehicle:FindFirstChildWhichIsA("BasePart", true)
    if not root then return end
    local dir = Vector3.zero
    local camCF = Camera.CFrame
    if flyControls.Forward  then dir += camCF.LookVector  end
    if flyControls.Backward then dir -= camCF.LookVector  end
    if flyControls.Left     then dir -= camCF.RightVector end
    if flyControls.Right    then dir += camCF.RightVector end
    if flyControls.Up       then dir += Vector3.yAxis     end
    if flyControls.Down     then dir -= Vector3.yAxis     end
    bv.Velocity = (dir.Magnitude > 0 and dir.Unit or Vector3.zero) * flySpeedVehicle
    bg.CFrame = camCF
end)

    bg = Instance.new("BodyGyro", root); bg.MaxTorque = Vector3.new(1e9,1e9,1e9); bg.P = 15000
end

local function LimparESPDoPlayer(player)
    local tbls = {ESP_Nomes, ESP_Tools, ESP_Skeletons, ESP_HealthBars, ESP_Tracers, ESP_Ranks}
    for _, tbl in ipairs(tbls) do
        if tbl[player] then
            for _, obj in pairs(tbl[player]) do
                if type(obj) == "table" then
                    for _, subObj in pairs(obj) do pcall(function() subObj.Visible = false subObj:Remove() end) end
                else
                    pcall(function() obj.Visible = false obj:Remove() end)
                end
            end
            tbl[player] = nil
        end
    end
end

local function UpdateNomes()
    if not ESP_Nome_Ativado then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local char = p.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChild("Humanoid")

        if vis then d.Text.Position = Vector2.new(s.X, s.Y-28); d.Text.Visible = true

local function UpdateSkeleton()
    if not ESP_Skeleton_Ativado then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local char = p.Character
        if not char then continue end
        local hum = char:FindFirstChild("Humanoid")
        if not hum or hum.Health <= 0 then LimparESPDoPlayer(p) continue end
        local head  = char:FindFirstChild("Head")
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        local lArm  = char:FindFirstChild("Left Arm")  or char:FindFirstChild("LeftUpperArm")
        local rArm  = char:FindFirstChild("Right Arm") or char:FindFirstChild("RightUpperArm")
        local lLeg  = char:FindFirstChild("Left Leg")  or char:FindFirstChild("LeftUpperLeg")
        local rLeg  = char:FindFirstChild("Right Leg") or char:FindFirstChild("RightUpperLeg")
        if not (head and torso) then continue end
        if not ESP_Skeletons[p] then
            ESP_Skeletons[p] = {}
            for _, c in ipairs({"Head-Torso","Torso-LArm","Torso-RArm","Torso-LLeg","Torso-RLeg"}) do
                local l = Drawing.new("Line"); l.Thickness = 3.0; l.Transparency = 0.9; l.Visible = false
                ESP_Skeletons[p][c] = l
            end
        end
        local lines = ESP_Skeletons[p]
        local cor = GetPlayerColor(p)
        local function draw(a, b, key)
            if not a or not b then if lines[key] then lines[key].Visible = false end return end
            local pa, va = Camera:WorldToViewportPoint(a.Position)
            local pb, vb = Camera:WorldToViewportPoint(b.Position)
            local l = lines[key]
            if va and vb then l.From, l.To, l.Color, l.Visible = Vector2.new(pa.X,pa.Y), Vector2.new(pb.X,pb.Y), cor, true
            else l.Visible = false end
        end
        draw(head,torso,"Head-Torso"); draw(torso,lArm,"Torso-LArm"); draw(torso,rArm,"Torso-RArm")
        draw(torso,lLeg,"Torso-LLeg"); draw(torso,rLeg,"Torso-RLeg")
    end
end

local function UpdateHealthBar()
    if not ESP_HealthBar_Ativado then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local char = p.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChild("Humanoid")
        local head = char and char:FindFirstChild("Head")
        if not char or not root or not hum or not head or hum.Health <= 0 then LimparESPDoPlayer(p) continue end
        if not ESP_HealthBars[p] then
            local bg2    = Drawing.new("Square"); bg2.Filled = true; bg2.Color = Color3.fromRGB(0,0,0)
            bg2.Transparency = 1; bg2.Visible = false; bg2.ZIndex = 1
            local health = Drawing.new("Square"); health.Filled = true; health.Transparency = 1; health.Visible = false; health.ZIndex = 2
            local text   = Drawing.new("Text"); text.Size = 14; text.Center = true; text.Outline = true
            text.OutlineColor = Color3.fromRGB(0,0,0); text.Visible = false; text.ZIndex = 3
            ESP_HealthBars[p] = {Bg = bg2, Bar = health, Text = text}
        end
        local d = ESP_HealthBars[p]
        local headScreen, headVisible = Camera:WorldToViewportPoint(head.Position + Vector3.new(0,0.85,0))
        local footScreen, footVisible = Camera:WorldToViewportPoint(root.Position + Vector3.new(0,-3.1,0))
        if headVisible and footVisible then
            local barHeight = math.floor(math.abs(headScreen.Y - footScreen.Y))
            local barWidth  = 3
            local pct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            local hpColor = Color3.fromRGB(0,255,0)
            if pct < 0.35 then hpColor = Color3.fromRGB(255,0,0)
            elseif pct < 0.75 then hpColor = Color3.fromRGB(255,255,0) end
            local xOffset  = barHeight * 0.40
            local barPosX  = math.floor(headScreen.X - xOffset)
            local barPosY  = math.floor(headScreen.Y)
            d.Bg.Size = Vector2.new(barWidth+2, barHeight+2); d.Bg.Position = Vector2.new(barPosX-1, barPosY-1); d.Bg.Visible = true
            local fillHeight = math.floor(barHeight * pct)
            d.Bar.Size = Vector2.new(barWidth, math.max(1, fillHeight))
            d.Bar.Position = Vector2.new(barPosX, barPosY + (barHeight - fillHeight))
            d.Bar.Color = hpColor; d.Bar.Visible = true
            d.Text.Text = math.floor(pct*100) .. "%"; d.Text.Color = hpColor
            d.Text.Position = Vector2.new(barPosX + (barWidth/2), barPosY + barHeight + 4); d.Text.Visible = true
        else d.Bg.Visible = false; d.Bar.Visible = false; d.Text.Visible = false end
    end
end

local function IniciarUpdates()
    local funcs = {
        {UpdateNomes,    "ESP_Nome_Ativado",      function() return ESP_Nome_Ativado end,      ESP_Nomes},
        {UpdateTools,    "ESP_Tool_Ativado",      function() return ESP_Tool_Ativado end,      ESP_Tools},

        {UpdateHealthBar,"ESP_HealthBar_Ativado", function() return ESP_HealthBar_Ativado end, ESP_HealthBars},
        {UpdateTracer,   "ESP_Tracer_Ativado",    function() return ESP_Tracer_Ativado end,    ESP_Tracers},
        {UpdateRank,     "ESP_Rank_Ativado",      function() return ESP_Rank_Ativado end,      ESP_Ranks},
    }
    for _, item in ipairs(funcs) do
        local func, key, getActive, tbl = item[1], item[2], item[3], item[4]
        if getActive() then
            if not Conns[key] then Conns[key] = RunService.RenderStepped:Connect(func) end
        else
            if Conns[key] then
                Conns[key]:Disconnect(); Conns[key] = nil
                for player, _ in pairs(tbl) do LimparESPDoPlayer(player) end
            end
        end
    end
end

local function CheckSafeZone(alvo)
    if not alvo or not alvo.Character then return false end
    if alvo.Character:FindFirstChildOfClass("ForceField") then return true end
    if alvo:GetAttribute("SafeZone") or alvo.Character:GetAttribute("SafeZone") then return true end
    return false
end

local function EstaVisivel(alvo)
    local pecas = Camera:GetPartsObscuringTarget({Camera.CFrame.Position, alvo.Position}, {LocalPlayer.Character, alvo.Parent})
    return #pecas == 0
end

local function GetClosestTarget()
    local target = nil
    local bestScore = math.huge
    local shortestDist = FOVRadius
    local Center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local prioTarget = nil
    local prioScore = math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and not table.find(IgnoredPlayers, p.Name) and not (IgnorarADM and IsStaffPlayer(p)) then
            if SafezoneCheckAimbot and CheckSafeZone(p) then continue end
            local part = p.Character:FindFirstChild(TargetPart)
            local hum  = p.Character:FindFirstChild("Humanoid")
            if part and hum and (not KillCheck or hum.Health > 0) then
                if TeamCheck and p.Team == LocalPlayer.Team then continue end
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    if WallCheck and not EstaVisivel(part) then continue end
                    local magnitude = (Vector2.new(pos.X,pos.Y) - Center).Magnitude
                    if magnitude > FOVRadius then continue end
                    local score = magnitude
                    if PriorizarMenosVida and not PriorizarMaisPerto then
                        score = hum.Health
                    elseif PriorizarMaisPerto and not PriorizarMenosVida then
                        score = (part.Position - Camera.CFrame.Position).Magnitude
                    elseif PriorizarMenosVida and PriorizarMaisPerto then
                        local worldDist = (part.Position - Camera.CFrame.Position).Magnitude
                        score = (hum.Health / math.max(hum.MaxHealth, 1)) * worldDist
                    end
                    if table.find(PriorizarJogadores, p.Name) then
                        if score < prioScore then prioTarget = part; prioScore = score end
                    else
                        if score < bestScore then target = part; bestScore = score; shortestDist = magnitude end
                    end
                end
            end
        end
    end
    return prioTarget or target
end

local function GetSilentTarget(targetPartName)
    targetPartName = targetPartName or TargetPart
    local char = LocalPlayer.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local Center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local target = nil
    local prioTarget = nil
    local bestScore = math.huge
    local prioScore = math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if not p.Character then continue end
        if table.find(IgnoredPlayers, p.Name) then continue end
        if IgnorarADM and IsStaffPlayer(p) then continue end
        if TeamCheck and p.Team == LocalPlayer.Team then continue end
        if SafezoneCheckSilentAim and CheckSafeZone(p) then continue end
        local part = p.Character:FindFirstChild(targetPartName)
        local hum  = p.Character:FindFirstChild("Humanoid")
        if not part or not hum then continue end
        if KillCheck and hum.Health <= 0 then continue end
        local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen then continue end
        if WallCheck and not EstaVisivel(part) then continue end
        local magnitude = (Vector2.new(pos.X, pos.Y) - Center).Magnitude
        if magnitude > FOVRadius then continue end
        local score = magnitude
        if PriorizarMenosVida and not PriorizarMaisPerto then
            score = hum.Health
        elseif PriorizarMaisPerto and not PriorizarMenosVida then
            score = (part.Position - Camera.CFrame.Position).Magnitude
        elseif PriorizarMenosVida and PriorizarMaisPerto then
            local worldDist = (part.Position - Camera.CFrame.Position).Magnitude
            score = (hum.Health / math.max(hum.MaxHealth, 1)) * worldDist
        end
        if table.find(PriorizarJogadores, p.Name) then
            if score < prioScore then prioTarget = part; prioScore = score end
        else
            if score < bestScore then target = part; bestScore = score end
        end
    end
    return prioTarget or target
end

local success_sa, result_sa = pcall(function()
    local RS = game:GetService("ReplicatedStorage")
    local OrpheusShared = RS:WaitForChild("OrpheusShared", 10)
    if not OrpheusShared then return end
    local Framework = OrpheusShared:WaitForChild("GunSys", 10)
    and OrpheusShared.GunSys:WaitForChild("Framework", 10)

    local WRC = Framework:WaitForChild("WeaponRaycastController", 10)
    if not WRC then return end
    local a = require(WRC)
    local b = clonefunction(a.PerformRaycast)
    a.PerformRaycast = newcclosure(function(c, d, e)
        if not SilentAimEnabled then return b(c, d, e) end
        
        if math.random(1, 100) > SilentHitChance then
            return b(c, d, e)
        end
        
        local effectivePart = TargetPart

            if TirosDisparados < TirosAntesDaCabeca then
                effectivePart = "HumanoidRootPart"
            else
                effectivePart = "Head"

            TirosDisparados = TirosDisparados + 1
            if TirosDisparados >= 30 then
                TirosDisparados = 0
            end
        end
        
        local h = GetSilentTarget(effectivePart)
        if not h then return b(c, d, e) end
        local f = LocalPlayer.Character
        local g = f and f:FindFirstChild("HumanoidRootPart")
        if not g then return b(c, d, e) end
        local n = c.Config and c.Config.Spread or 0
        return h, h.Position,
        (g.Position - h.Position).Unit,
        (CFrame.new(workspace.CurrentCamera.Focus.Position,
        workspace.CurrentCamera.CFrame.LookVector * 500)
        * CFrame.Angles(
        math.rad(-n + math.random() * (n * 2)),

    end)

    if p.Team and p.Team.Name == "STAFF" then return true end
    if table.find(StaffList, p.Name) then return true end
    local ok, rank = pcall(function() return p:GetRoleInGroup(GrupoSintoniaID) end)
    if ok and rank and rank:lower():find("staff") then return true end
    local ok2, rankN = pcall(function() return p:GetRankInGroup(GrupoSintoniaID) end)

local XRayAtivado = false
local XRayOriginals = {}

local function AplicarXRay(ativo)
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("MeshPart") or obj:IsA("UnionOperation") then
            if not obj:IsDescendantOf(LocalPlayer.Character) then
                local isCharPart = false
                for _, p in pairs(Players:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character and obj:IsDescendantOf(p.Character) then
                        isCharPart = true; break
                    end
                end
                if not isCharPart then
                    if ativo then
                        if not XRayOriginals[obj] then
                            XRayOriginals[obj] = { Transparency = obj.Transparency, CastShadow = obj.CastShadow }
                        end
                        obj.Transparency = 0.9
                        obj.CastShadow = false
                    else
                        if XRayOriginals[obj] then
                            pcall(function()
                                obj.Transparency = XRayOriginals[obj].Transparency
                                obj.CastShadow = XRayOriginals[obj].CastShadow
                            end)
                        end
                    end
                end
            end
        end
    end
    if not ativo then XRayOriginals = {} end
end

GPlayers:AddToggle("MapaTransparente", {
    Text = "Mapa Transparente", Default = false,
    Callback = function(Value)
        XRayAtivado = Value
        AplicarXRay(Value)
    end,

local flySpeed = 35
local isNormalFlyEnabled = false
local isGhostFlyEnabled = false
local flyModifiedParts = {}

local ghostActive = false
local ghostConnection = nil
local fakeCharacter = nil
local fakeRoot = nil
local fakeHumanoid = nil

local platform = Instance.new("Part")
platform.Size = Vector3.new(6, 1, 6)
platform.Transparency = 1 
platform.Anchored = true
platform.CanCollide = false
platform.Name = "SkyFly_Platform"

local overlapParams = OverlapParams.new()
overlapParams.FilterType = Enum.RaycastFilterType.Exclude

local function ActivateGhost()
    local char = LocalPlayer.Character
    if not char then return end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChild("Humanoid")
    if not root or not humanoid then return end
    
    ghostActive = true
    
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            part.Transparency = 1
            part.CanCollide = true
        end
    end
    
    humanoid.PlatformStand = false
    humanoid.WalkSpeed = 16
    root.Anchored = false
    
    if fakeCharacter then
        fakeCharacter:Destroy()
        fakeCharacter = nil
    end
    
    fakeCharacter = Instance.new("Model")
    fakeCharacter.Name = "Fake_" .. LocalPlayer.Name
    fakeCharacter.Parent = workspace
    
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            local clone = part:Clone()
            clone.Parent = fakeCharacter
            clone.CanCollide = false
            clone.Transparency = 0
            clone.Anchored = false
        end
    end
    
    fakeHumanoid = Instance.new("Humanoid")
    fakeHumanoid.Parent = fakeCharacter
    fakeHumanoid.PlatformStand = true
    fakeHumanoid.WalkSpeed = humanoid.WalkSpeed
    
    fakeRoot = fakeCharacter:FindFirstChild("HumanoidRootPart")
    if fakeRoot then
        fakeRoot.CFrame = root.CFrame
    end
    
    workspace.CurrentCamera.CameraSubject = fakeHumanoid
    
    if ghostConnection then
        ghostConnection:Disconnect()
        ghostConnection = nil
    end
    
    ghostConnection = RunService.Heartbeat:Connect(function()
        if not ghostActive then return end
        
        local currentChar = LocalPlayer.Character
        if not currentChar then return end
        
        local currentRoot = currentChar:FindFirstChild("HumanoidRootPart")
        if not currentRoot then return end
        
        for _, part in ipairs(currentChar:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Transparency = 1
                part.CanCollide = true
            end
        end
        
        if fakeRoot and fakeRoot.Parent then
            local realPos = currentRoot.Position
            fakeRoot.CFrame = CFrame.new(realPos)
            fakeRoot.Velocity = currentRoot.Velocity
            fakeRoot.CFrame = CFrame.lookAt(realPos, realPos + currentRoot.CFrame.LookVector)
        end
    end)
end

local function DeactivateGhost()
    ghostActive = false
    if ghostConnection then
        ghostConnection:Disconnect()
        ghostConnection = nil
    end
    if fakeCharacter then
        fakeCharacter:Destroy()
        fakeCharacter = nil
        fakeRoot = nil
        fakeHumanoid = nil
    end
    local char = LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Transparency = 0
                part.CanCollide = true
            end
        end
    end
    if char then
        local hum = char:FindFirstChild("Humanoid")
        if hum then workspace.CurrentCamera.CameraSubject = hum end
    end
end

local function UpdateFlyState()
    local activeFly = isNormalFlyEnabled or isGhostFlyEnabled
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if activeFly then
        if hrp then
            platform.CFrame = hrp.CFrame * CFrame.new(0, -3.1, 0)
            platform.CanCollide = true
            platform.Parent = workspace
        end
    else
        platform.CanCollide = false
        platform.Parent = nil
        if hum then 
            hum.WalkSpeed = 16
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        if hrp then
            hrp.Velocity = Vector3.zero
            hrp.RotVelocity = Vector3.zero
        end
        for part, _ in pairs(flyModifiedParts) do
            if part and part.Parent then part.CanCollide = true end
        end
        table.clear(flyModifiedParts)
    end
end

RunService.RenderStepped:Connect(function(deltaTime)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local activeFly = isNormalFlyEnabled or isGhostFlyEnabled
    
    if activeFly and hrp and hum then
        local camera = workspace.CurrentCamera
        hum:ChangeState(Enum.HumanoidStateType.Swimming)
        hum.WalkSpeed = 16
        local moveDir = hum.MoveDirection
        local lookVec = camera.CFrame.LookVector
        
        if moveDir.Magnitude > 0 then
            local isMovingBackward = moveDir:Dot(camera.CFrame.LookVector.Unit) < -0.5
            local verticalMove = lookVec.Y
            if isMovingBackward then verticalMove = -verticalMove end
            
            local smoothedDir = Vector3.new(moveDir.X, verticalMove, moveDir.Z).Unit
            local targetCFrame = hrp.CFrame + (smoothedDir * flySpeed * deltaTime)
            
            hrp.CFrame = CFrame.lookAt(targetCFrame.Position, targetCFrame.Position + Vector3.new(lookVec.X, 0, lookVec.Z))
            platform.CFrame = targetCFrame * CFrame.new(0, -3.1, 0)
            hrp.Velocity = Vector3.zero 
            hrp.RotVelocity = Vector3.zero
        else
            hrp.Velocity = Vector3.zero
            hrp.RotVelocity = Vector3.zero
            platform.CFrame = CFrame.new(hrp.Position.X, hrp.Position.Y - 3.1, hrp.Position.Z)
        end
        
        overlapParams.FilterDescendantsInstances = {char, platform}
        local parts = workspace:GetPartBoundsInBox(hrp.CFrame, Vector3.new(4, 5, 4), overlapParams)
        local currentFrameParts = {}
        
        for _, part in ipairs(parts) do
            if part:IsA("BasePart") then
                if part.CanCollide then
                    flyModifiedParts[part] = true
                    part.CanCollide = false
                end
                currentFrameParts[part] = true
            end
        end
        
        for part, _ in pairs(flyModifiedParts) do
            if not currentFrameParts[part] then
                if part and part.Parent then part.CanCollide = true end
                flyModifiedParts[part] = nil
            end
        end
    else
        if platform.Parent then platform.Parent = nil end
    end
end)

local noclipActive = false

                        if part.CanCollide then
                            noclipModifiedParts[part] = true
                            part.CanCollide = false
                        end
                        currentFrameNoclip[part] = true
                    end
                end
            end
            
            for part, _ in pairs(noclipModifiedParts) do
                if not currentFrameNoclip[part] then
                    if part and part.Parent then part.CanCollide = true end
                    noclipModifiedParts[part] = nil
                end
            end
        end
    end
end)

local function TeleportarSeguro(posX, posY, posZ)
    local char = LocalPlayer.Character
    if not char then return end
    local rootPart = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not rootPart or not humanoid then return end
    
    local objetosSalvos = {}
    rootPart.Velocity = Vector3.new(0, 55, 0)
    task.wait(0.1)
    
    humanoid:ChangeState(Enum.HumanoidStateType.Physics)
    rootPart.Velocity = Vector3.new(0, 0, 0)
    
    for _, objeto in ipairs(workspace:GetChildren()) do
        if objeto ~= char and objeto ~= workspace.CurrentCamera and 
           not objeto:IsA("Terrain") and not objeto:IsA("Script") and
           not objeto:IsA("ModuleScript") then
            pcall(function()
                objetosSalvos[objeto] = objeto.Parent
                objeto.Parent = nil
            end)
        end
    end
    
    rootPart.CFrame = CFrame.new(posX, posY, posZ)
    char:MoveTo(Vector3.new(posX, posY, posZ))
    task.wait(0.1)
    
    for objeto, parentOriginal in pairs(objetosSalvos) do
        pcall(function() objeto.Parent = parentOriginal end)
    end
    
    rootPart.Velocity = Vector3.new(0, 0, 0)
    rootPart.RotVelocity = Vector3.new(0, 0, 0)
    task.wait(0.08)
    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    task.wait(0.05)
    humanoid:ChangeState(Enum.HumanoidStateType.Standing)
    
    if LocalPlayer.RequestStreamAroundAsync then
        pcall(function() LocalPlayer:RequestStreamAroundAsync(rootPart.Position) end)
    end
end

local function CarroEstaTrancado(veiculo)
    if veiculo:GetAttribute("Locked")    == true then return true end
    if veiculo:GetAttribute("Trancado")  == true then return true end
    if veiculo:GetAttribute("IsLocked")  == true then return true end
    if veiculo:GetAttribute("locked")    == true then return true end
    local lockVal = veiculo:FindFirstChild("Locked") or veiculo:FindFirstChild("Trancado") or veiculo:FindFirstChild("IsLocked")
    if lockVal and lockVal:IsA("BoolValue") and lockVal.Value == true then return true end
    if veiculo:FindFirstChild("LockPart") then return true end
    return false
end

local function DetectarCarros()
    ListaCarros = {}
    local nomes        = {}
    local nomesVistos  = {}
    for _, obj in pairs(workspace:GetDescendants()) do
        local seat = nil
        if obj:IsA("VehicleSeat") then
            seat = obj
        elseif obj.Name == "DriveSeat" and obj:IsA("BasePart") then
            seat = obj
        end
        if seat then
            if seat.Occupant ~= nil then continue end
            local veiculo     = seat.Parent
            local nomeVeiculo = veiculo.Name
            if CarroEstaTrancado(veiculo) then continue end
            if nomesVistos[nomeVeiculo] then continue end
            nomesVistos[nomeVeiculo] = true
            table.insert(ListaCarros, { nome = nomeVeiculo, seat = seat, veiculo = veiculo })
            table.insert(nomes, nomeVeiculo)
        end
    end
    if #nomes == 0 then
        CarrosDropdown:SetValues({"Nenhum"})
        CarrosDropdown:SetValue("Nenhum")
        CarroSelecionado = nil
    else
        CarrosDropdown:SetValues(nomes)
        local mantemSelecao = false
        if CarroSelecionado then
            for _, n in ipairs(nomes) do
                if n == CarroSelecionado.nome then mantemSelecao = true; break end
            end
        end
        if not mantemSelecao then
            CarrosDropdown:SetValue(nomes[1])
            CarroSelecionado = ListaCarros[1]
        end
    end
end

local function PuxarEEntrarCarro(dados)
    if not dados then return false end
    local seat    = dados.seat
    local veiculo = dados.veiculo
    if not seat or not seat.Parent then return false end
    if seat.Occupant ~= nil then return false end
    if veiculo and CarroEstaTrancado(veiculo) then return false end
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return false end
    local primaryPart = veiculo.PrimaryPart or veiculo:FindFirstChildWhichIsA("BasePart")
    if primaryPart then
        local offsetPos = root.Position + Vector3.new(0, 0, 0)
        veiculo:SetPrimaryPartCFrame(CFrame.new(offsetPos + Vector3.new(0, 3, 0)))
        task.wait(0.15)
    end
    if seat.Occupant ~= nil then return false end
    if seat:IsA("VehicleSeat") then
        seat:Sit(hum)
    else
        hum.SeatPart = seat
    end
    return true
end
